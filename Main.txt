// Universal StoneScript auto script.
// Equipment names are candidates, not fixed selections: the first item owned is used.

var locations = ["Rocky Plateau", "Deadwood Canyon", "Caves of Fear", "Mushroom Forest", "Haunted Halls", "Boiling Mine", "Icy Ridge", "Temple"]
var locationPotions = ["defensive", "vampiric", "lucky", "vampiric", "vampiric", "strength", "lucky", "berserk"]
var elements = ["none", "vigor", "aether", "fire", "ice", "poison"]
var counterElements = ["none", "poison", "vigor", "aether", "fire", "ice"]
var ranges = ["melee", "close", "medium", "ranged", "screen"]
var rangeUnits = [5, 11, 16, 25, 35]
var healingPotions = ["defensive", "vampiric", "cleansing", "healing"]
var offensivePotions = ["strength", "lightning", "berserk", "lucky"]

// Ordered from preferred to fallback.  GetOwned() ignores anything not in inventory.
var meleeWeapons = ["big aether sword", "big vigor sword", "poison hammer", "vigor hammer", "fire hammer", "ice hammer", "ice sword", "heavy hammer", "bardiche"]
var rangedWeapons = ["aether staff", "ice staff", "vigor staff", "fire talisman", "aether talisman"]
var physicalRangedWeapons = ["ice staff", "aether staff", "vigor staff"]
var shields = ["aether shield", "towering shield", "fire shield", "poison shield", "dashing shield", "bashing shield", "shield"]
var dashShields = ["dashing shield", "bashing shield"]
var armorBreakWeapons = ["heavy hammer"]
var splashWeapons = ["big aether sword", "bardiche", "big vigor sword"]
var debuffPoison = ["poison sword dP"]
var debuffFire = ["fire sword dF"]
var debuffIce = ["ice sword dI"]

var currentLocation
var currentPotion
var counterElement
var chosenMelee
var chosenRanged
var chosenShield
var chosenOffhand
var chosenBoth
var queuedActivation
var dashBlocked
var mindFreezeUntil

func GetOwned(candidates)
	for candidate : candidates
		?item.GetCount(candidate) > 0
			return candidate
	return false

func Has(itemName)
	return itemName & item.GetCount(itemName) > 0

func IsHealing(potion)
	return potion & healingPotions.Contains(potion)

func IsOffensive(potion)
	return potion & offensivePotions.Contains(potion)

func Lookup(value, source, result)
	var index = source.IndexOf(value)
	?index > -1
		return result[index]
	return false

func InRange(minimum, maximum)
	?foe.count > 0
		?maximum > minimum
			return foe.distance >= minimum & foe.distance <= maximum
		return foe.distance >= minimum
	return false

func CloseAs(rangeName)
	return foe.distance <= Lookup(rangeName, ranges, rangeUnits)

func FarAs(rangeName)
	return foe.distance >= Lookup(rangeName, ranges, rangeUnits)

func CounterElement()
	?foe.count > 0
		for i = 1..5
			var element = elements[i]
			var counter = counterElements[i]
			var defenseBuff = "adaptive_defense_" + counter
			?foe = element & foe.buffs.string ! defenseBuff
				return counter
	return false

func CounterWeapon(counter)
	?counter = "poison"
		return GetOwned(["poison hammer", "poison sword dP", "poison sword"])
	?counter = "vigor"
		return GetOwned(["vigor hammer", "vigor sword", "vigor staff"])
	?counter = "aether"
		return GetOwned(["aether hammer", "aether sword", "big aether sword"])
	?counter = "fire"
		return GetOwned(["fire hammer", "fire sword", "fire talisman"])
	?counter = "ice"
		return GetOwned(["ice hammer", "ice sword", "ice staff"])
	return false

func IsTwoHanded(itemName)
	return itemName = "big aether sword" | itemName = "big vigor sword" | itemName = "heavy hammer" | itemName = "bardiche" | itemName = "aether staff" | itemName = "ice staff" | itemName = "vigor staff" | itemName = "ice wand"

func EquipLeft(itemName)
	?Has(itemName) & !chosenOffhand
		chosenOffhand = itemName

func EquipRight(itemName)
	?Has(itemName) & !chosenShield
		chosenShield = itemName

func EquipBoth(itemName)
	?Has(itemName)
		chosenBoth = itemName

func QueueActivate(itemName)
	?itemName & Has(itemName)
		queuedActivation = itemName

// Attack-animation cancelling (the frame-skip glitch). The dummy items interrupt
// the post-hit animation, then the normal decision code restores the loadout.
func CheckForAAC()
	?ai.enabled
		?item.left.state = 3
			equipL wand *0
		?item.right.state = 3
			equipR shield *0

func ChooseEquipment()
	chosenMelee = false
	chosenRanged = false
	chosenShield = false
	chosenOffhand = false
	chosenBoth = false
	queuedActivation = false

	counterElement = CounterElement()
	chosenMelee = CounterWeapon(counterElement)
	?!chosenMelee
		chosenMelee = GetOwned(meleeWeapons)
	chosenRanged = GetOwned(rangedWeapons)
	chosenShield = GetOwned(shields)

	?foe.count > 3 & CloseAs("melee")
		var splash = GetOwned(splashWeapons)
		?splash
			chosenMelee = splash
	?foe.armor > 0 & CloseAs("melee")
		var armorBreaker = GetOwned(armorBreakWeapons)
		?armorBreaker
			chosenMelee = armorBreaker

	// Physical immunity requires a staff; ranged fallback is still preferable at distance.
	?foe = immune_to_physical
		chosenMelee = GetOwned(physicalRangedWeapons)
	?foe = immune_to_ranged
		chosenRanged = false
	?foe = swarm | foe = insect
		chosenMelee = false
		chosenRanged = GetOwned(rangedWeapons)
		?!chosenRanged
			chosenMelee = GetOwned(meleeWeapons)

	?foe.distance > 11 & foe.distance <= 25 & chosenRanged
		chosenMelee = false
	?foe.distance > 25
		chosenMelee = false
	?!chosenMelee & !chosenRanged & foe.count > 0
		// Never enter combat with only a shield: use the strongest owned melee fallback.
		chosenMelee = GetOwned(meleeWeapons)

	?foe.buffs.string ! protection & CloseAs("close") & buffs.string ! berserk & buffs.string ! lucky & buffs.string ! strength
		?foe.debuffs.string ! damage
			var poison = GetOwned(debuffPoison)
			?poison
				chosenMelee = poison
		:?foe.debuffs.string ! dot
			var fire = GetOwned(debuffFire)
			?fire
				chosenMelee = fire
		:?foe.debuffs.string ! chill
			var ice = GetOwned(debuffIce)
			?ice
				chosenMelee = ice
		:?foe.debuffs.string ! feeble & (Has("cult mask") | Has("mask")) & item.GetCooldown("mask") <= 0
			?Has("cult mask")
				chosenShield = "cult mask"
				QueueActivate("cult mask")
			:
				chosenShield = "mask"
				QueueActivate("mask")

	?foe.count > 0 & CloseAs("close") & item.GetCooldown("bash") <= 0 & !dashBlocked
		?foe.count > 5 | foe.damage > 4 | foe = boss
			var bash = GetOwned(["bashing shield *7", "bashing shield"])
			?bash
				chosenShield = bash
				QueueActivate(bash)

	?foe.count > 0 & foe.distance >= 9 & foe.distance <= 14 & item.GetCooldown("dash") <= 0 & !dashBlocked
		var dash = GetOwned(dashShields)
		?dash
			chosenShield = dash
			QueueActivate(dash)

	?foe.count > 0 & CloseAs("melee") & Has("skeleton arm") & item.GetCooldown("skeleton_arm") <= 0 & item.CanActivate()
	^& foe ! immune_to_physical & buffs.string ! "pick_pocket:3"
		chosenBoth = "skeleton arm"
		?foe.hp + foe.armor < 10 | foe.count > 3
			QueueActivate("skeleton arm")

	?foe.count > 5 & CloseAs("ranged") & Has("blade of the fallen god") & item.GetCooldown("blade") <= 0 & item.CanActivate()
		chosenBoth = "blade of the fallen god"
		QueueActivate("blade of the fallen god")

	// Mind dodges are location-specific and intentionally conservative.
	?item.GetCooldown("mind") <= 0
		?foe = dysangelos & (foe.state = 33 | foe.state = 115)
			EquipBoth("mind")
			mindFreezeUntil = totalTime + 60
		:?foe = bronze_guardian & foe.state = 32 & foe.time >= 10
			EquipBoth("mind")
			mindFreezeUntil = totalTime + 60
		:?foe = nagaraja & foe.state = 112 & foe.time >= 53
			EquipBoth("mind")
			mindFreezeUntil = totalTime + 60

	?!foe.count | foe.distance > 25
		// Star Stone magnet and Triskelion movement bonus occupy separate hands.
		// Assign both every frame so stale combat or pickup gear is replaced.
		chosenMelee = false
		chosenOffhand = GetOwned(["star stone"])
		chosenShield = GetOwned(["triskelion"])
		?hp < maxhp
			chosenOffhand = GetOwned(["ouroboros", "star stone"])
		:?harvest = Boulder
			chosenBoth = GetOwned(["shovel"])
		:?harvest = Tree
			chosenShield = GetOwned(["triskelion"])
		:?pickup & pickup.distance < 13
			chosenOffhand = GetOwned(["star stone", "star"])

func UsePotion()
	?item.potion ! empty
		?IsHealing(item.potion) & hp < (0.35 * maxhp)
			activate potion
		:?IsOffensive(item.potion) & foe.count > 0 & (foe = boss | foe = dysangelos | foe = bolesh | foe = yeti) & foe.state > 1
			activate potion
		:?item.potion = cleansing & debuffs.count > 15
			activate potion

func AreaRules()
	// Area checker: these boss rules run only in the named area.
	?loc = rocky
		?foe = dysangelos | foe = acronian
			dashBlocked = true
	:?loc = deadwood
		?foe = poena | foe = xyloalgia
			dashBlocked = true
	:?loc = caves
		?foe = bolesh | foe = ceiling
			dashBlocked = true
	:?loc = fungus
		?foe = morel
			dashBlocked = true
	:?loc = undead
		?foe = pallas
			dashBlocked = true
	:?loc = mine
		?foe = bronze_guardian | foe = guardian
			dashBlocked = true
	:?loc = ridge
		?foe = yeti | foe = ice wall | foe = ice pillar
			dashBlocked = true
	:?loc = temple
		?foe = nagaraja | foe = acronian
			dashBlocked = true

func EquipFinal()
	?chosenBoth
		equip @chosenBoth@
	:?chosenMelee
		?IsTwoHanded(chosenMelee)
			equip @chosenMelee@
		:
			equipL @chosenMelee@
	?chosenRanged & !chosenMelee & !chosenBoth
		?IsTwoHanded(chosenRanged)
			equip @chosenRanged@
		:
			equipR @chosenRanged@
	?chosenOffhand & !chosenMelee & !chosenBoth
		equipL @chosenOffhand@
	?chosenShield & !chosenBoth & !chosenRanged & !IsTwoHanded(chosenMelee)
		equipR @chosenShield@
	?queuedActivation & item.CanActivate()
		activate R

?loc.begin
	currentLocation = loc.name
	currentPotion = item.potion

?Type(mindFreezeUntil) = int & totalTime >= mindFreezeUntil
	mindFreezeUntil = false
dashBlocked = false

CheckForAAC()
AreaRules()
UsePotion()
ChooseEquipment()

?foe.count > 0 & foe.distance > 25 & summon.count = 0 & item.GetCooldown("talisman_fire") <= 0 & item.CanActivate()
	?Has("fire talisman")
		equipR fire talisman
		activate R

EquipFinal()
