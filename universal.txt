  equipr bash

^＂bardiche＂)<1|item.GetCooldown(＂bardiche＂)>870)

func hsmash() //hammer special
 ?foe.distance<=8&(item.GetCooldown(
 ^＂hammer＂)<1|item.GetCooldown(＂hammer＂)>630)
 ^&foe.armor>1
  equip heavy hammer
  activate R

func ooc() //out of combat regen + shield
 ?foe.distance>22
  ?hp<maxhp
   equipL ouro
  :
   equipL triskelion
  ?element = ＂repeating＂
   equipR compound shield
  :
   string = element + ＂ shield max A golden＂
   equipR @string@

func steal() //skeleton arm steal
//adjust to arm level
 ?item.GetCooldown(＂skeleton_arm＂) <= 0
 ^&foe!une_to_physical^&foe.distance<10
  equip skeleton arm
  ?item.CanActivate(＂skeleton_arm＂)
  ^&buffs.string=skel_lvl
   activate R

func fireelem() //Summon Fire elemental
 ?summon.count=0&item.CanActivate()&time>30&
 ^foe.distance>30
  equipL fire talisman
  activate L

 ^|item.GetCooldown(＂mask＂)>2370)
 ^&foe.state=33
  equipR mask
  ?time%5=0
   activate R
